     1→/* ============================================================================
     2→ * INTRO VOICE — studio-grade, multi-language server-side speech.
     3→ * ----------------------------------------------------------------------------
     4→ * The side widget / hero button greets every visitor in a language they
     5→ * understand (visitor language detection happens client-side):
     6→ *   en — English        (native-quality male narrator)
     7→ *   ur — Urdu           (romanized script; accented but fully intelligible)
     8→ *   zh — Chinese        (native-quality)
     9→ * Other visitor languages fall back to English.
    10→ *
    11→ * Pipeline per language (all server-side, cached on disk):
    12→ *   1. Build a natural voice-over script from the editable SiteProfile
    13→ *      (numbers spelled out — TTS pronounces words better than digits).
    14→ *   2. Synthesize speech with the "xiaochen" voice (deep male narrator,
    15→ *      117 Hz — the only true male voice; the previous "jam" voice was
    16→ *      female-pitched AND mispronounced "Ubaid" as "Babe").
    17→ *   3. Master with ffmpeg: high-pass, mud-cut EQ, presence/air boost,
    18→ *      broadcast compression — then mix a subtle synthesized ambient music
    19→ *      bed underneath and loudness-normalize to -16 LUFS (YouTube/Meta
    20→ *      target), with a clean fade-out.
    21→ * A profile save invalidates every cached language.
    22→ * ========================================================================== */

    38→export function isVoiceLang(v: string | null | undefined): v is VoiceLang {

    42→/** Normalize a BCP-47 tag ("ar-AE", "en-GB", "pt-BR") to a voice language. */
    43→export function normalizeVoiceLang(raw: string | null | undefined): VoiceLang {
    44→  const base = (raw ?? "").trim().toLowerCase().split(/[-_]/)[0];
    45→  return isVoiceLang(base) ? base : "en";
    46→}

    67→/** 0..999999 → English words ("two hundred and fifty"). TTS-safe. */
    68→export function intToWords(n: number): string {
    69→  if (!Number.isFinite(n) || n < 0) return String(n);
    70→  const i = Math.round(n);
    71→  if (i < 20) return ONES[i];
    72→  if (i < 100) {
    73→    const t = Math.floor(i / 10);
    74→    const r = i % 10;
    75→    return r ? `${TENS[t]} ${ONES[r]}` : TENS[t];
    76→  }
    77→  if (i < 1000) {
    78→    const h = Math.floor(i / 100);
    79→    const r = i % 100;
    80→    return r ? `${ONES[h]} hundred and ${intToWords(r)}` : `${ONES[h]} hundred`;
    81→  }
    82→  const th = Math.floor(i / 1000);
    83→  const r = i % 1000;
    84→  const tail = r ? (r < 100 ? " and " : " ") + intToWords(r) : "";
    85→  return `${intToWords(th)} thousand${tail}`;
    86→}

   348→/**
   349→ * Write a warm ambient pad as a 48 kHz stereo WAV — five slow-evolving sine
   350→ * pairs with gentle detune (width), a soft attack and a long fade-out.
   351→ */
   352→function synthesizeBed(wavPath: string, seconds: number): void {
   353→  const SR = 48000;
   354→  const CH = 2;
   355→  const n = Math.max(1, Math.floor(seconds * SR));
   356→  // [frequency, relative amplitude] — A2, E3, A3, B3, E4 (Am add9, warm)
   357→  const NOTES: Array<[number, number]> = [
   358→    [110.0, 1.0],
   359→    [164.81, 0.72],
   360→    [220.0, 0.5],
   361→    [246.94, 0.4],
   362→    [329.63, 0.28],
   363→  ];
   364→  const l = new Float32Array(n);
   365→  const r = new Float32Array(n);
   366→  for (const [f, a] of NOTES) {
   367→    const fl = f * 0.9986;
   368→    const fr = f * 1.0014;
   369→    const pl = f * 1.7;
   370→    const pr = f * 3.1;
   371→    for (let i = 0; i < n; i++) {
   372→      const t = i / SR;
   373→      l[i] += a * Math.sin(2 * Math.PI * fl * t + pl);
   374→      r[i] += a * Math.sin(2 * Math.PI * fr * t + pr);
   375→    }
   376→  }
   377→  const ATTACK = 2.4;
   378→  const RELEASE = 4.0;
   379→  let peak = 1e-9;
   380→  for (let i = 0; i < n; i++) {
   381→    const t = i / SR;
   382→    let env = t < ATTACK ? (t / ATTACK) * (t / ATTACK) * (3 - 2 * (t / ATTACK)) : 1;
   383→    const relStart = Math.max(0, seconds - RELEASE);
   384→    if (t > relStart) {
   385→      const p = Math.min(1, (t - relStart) / RELEASE);
   386→      env *= 0.5 * (1 + Math.cos(Math.PI * p));
   387→    }
   388→    env *= 1 + 0.15 * Math.sin(2 * Math.PI * 0.13 * t);
   389→    env *= 1 + 0.12 * Math.sin(2 * Math.PI * 0.09 * t + 1.2);
   390→    l[i] *= env;
   391→    r[i] *= env;
   392→    const ap = Math.abs(l[i]);
   393→    const bp = Math.abs(r[i]);
   394→    if (ap > peak) peak = ap;
   395→    if (bp > peak) peak = bp;
   396→  }
   397→  const scale = BED_PEAK / peak;
   398→  const dataSize = n * CH * 2;
   399→  const buf = Buffer.alloc(44 + dataSize);
   400→  buf.write("RIFF", 0);
   401→  buf.writeUInt32LE(36 + dataSize, 4);
   402→  buf.write("WAVE", 8);
   403→  buf.write("fmt ", 12);
   404→  buf.writeUInt32LE(16, 16);
   405→  buf.writeUInt16LE(1, 20);
   406→  buf.writeUInt16LE(CH, 22);
   407→  buf.writeUInt32LE(SR, 24);
   408→  buf.writeUInt32LE(SR * CH * 2, 28);
   409→  buf.writeUInt16LE(CH * 2, 32);
   410→  buf.writeUInt16LE(16, 34);
   411→  buf.write("data", 36);
   412→  buf.writeUInt32LE(dataSize, 40);
   413→  for (let i = 0; i < n; i++) {
   414→    const lv = Math.max(-1, Math.min(1, l[i] * scale));
   415→    const rv = Math.max(-1, Math.min(1, r[i] * scale));
   416→    buf.writeInt16LE(Math.round(lv * 32767), 44 + i * 4);
   417→    buf.writeInt16LE(Math.round(rv * 32767), 44 + i * 4 + 2);
   418→  }
   419→  fs.writeFileSync(wavPath, buf);
   420→  fs.writeFileSync(wavPath.replace(/\.wav$/, ".duration.txt"), seconds.toFixed(2));
   421→}

   423→async function ensureBed(lang: VoiceLang, voiceSeconds: number): Promise<string> {
   424→  const bedPath = path.join(AUDIO_DIR, `intro-bed-${lang}.wav`);
   425→  const metaPath = bedPath.replace(/\.wav$/, ".duration.txt");
   426→  const need = voiceSeconds + 3.0; // voice + 2.2s pad + tail margin
   427→  try {
   428→    const cached = parseFloat(fs.readFileSync(metaPath, "utf8"));
   429→    if (Number.isFinite(cached) && Math.abs(cached - need) < 0.75 && fs.existsSync(bedPath)) {
   430→      return bedPath;
   431→    }
   432→  } catch {
   433→    /* no cache yet */
   434→  }
   435→  synthesizeBed(bedPath, need);
   436→  return bedPath;
   437→}

   478→      return bedPath;

   487→/**
   488→ * Master an admin-uploaded voice file with the exact same studio pipeline as
   489→ * the generated audio (EQ → compression → ambient bed → loudnorm → fade) and
   490→ * install it as the custom voice for the language.
   491→ */
   492→export async function saveCustomIntroVoice(lang: VoiceLang, inputPath: string): Promise<{ seconds: number }> {
   493→  const dur = probeDuration(inputPath);
   494→  if (!dur || dur < 3 || dur > 180) {
   495→    throw new Error("Audio must be between 3 and 180 seconds long.");
   496→  }
   497→  fs.mkdirSync(AUDIO_DIR, { recursive: true });
   498→  const bed = await ensureCustomBed(lang, dur);
   499→  const total = dur + 2.2;
   500→  const fadeStart = Math.max(0, total - 0.9);
   501→  const outPath = customPathFor(lang);
   502→  const tmpOut = outPath + ".tmp.mp3";
   503→  try {
   504→    execFileSync(
   505→      "ffmpeg",
   506→      [
   507→        "-y", "-v", "error",
   508→        "-i", inputPath,
   509→        "-i", bed,
   510→        "-filter_complex",
   511→        "[0:a]highpass=f=75," +
   512→          "equalizer=f=260:t=q:w=1.2:g=-2.5," +
   513→          "equalizer=f=3300:t=q:w=1.6:g=2.5," +
   514→          "equalizer=f=8500:t=q:w=2:g=1.8," +
   515→          "acompressor=threshold=-21dB:ratio=3:attack=8:release=140:makeup=3," +
   516→          "aresample=48000," +
   517→          "aformat=sample_fmts=fltp:channel_layouts=stereo," +
   518→          "apad=pad_dur=2.2[vc];" +
   519→          "[1:a]aformat=sample_fmts=fltp:channel_layouts=stereo[bc];" +
   520→          "[vc][bc]amix=inputs=2:duration=first:dropout_transition=0:normalize=0[mx];" +
   521→          `[mx]loudnorm=I=-16:TP=-1.5:LRA=11,aresample=48000,afade=t=out:st=${fadeStart.toFixed(2)}:d=0.9[out]`,
   522→        "-map", "[out]",
   523→        "-codec:a", "libmp3lame",
   524→        "-b:a", "160k",
   525→        "-ar", "48000",
   526→        tmpOut,
   527→      ],
   528→      { timeout: 120000 },
   529→    );
   530→    fs.renameSync(tmpOut, outPath);
   531→  } finally {
   532→    try { fs.unlinkSync(tmpOut); } catch { /* already renamed */ }
   533→  }
   534→  return { seconds: Math.round(dur * 10) / 10 };
   535→}

   541→function probeDuration(file: string): number {
   542→  const out = execFileSync(
   543→    "ffprobe",
   544→    ["-v", "error", "-show_entries", "format=duration", "-of", "default=nw=1:nk=1", file],
   545→    { timeout: 20000 },
   546→  )
   547→    .toString()
   548→    .trim();
   549→  return parseFloat(out) || 0;
   550→}

   552→/** TTS → EQ/compression → mix subtle bed → loudnorm (-16 LUFS) → MP3. */
   553→async function generateAudio(lang: VoiceLang, script: string): Promise<string> {
   554→  const ZAI = (await import("z-ai-web-dev-sdk")).default;
   555→  const zai = await ZAI.create();
   556→
   557→  // Retry with backoff — a throttled TTS API shouldn't fail the visitor's click.
   558→  let buffer: Buffer | null = null;
   559→  for (let attempt = 0; attempt < 4 && !buffer; attempt++) {
   560→    if (attempt > 0) await new Promise((r) => setTimeout(r, 15000 * attempt));
   561→    try {
   562→      const response = await zai.audio.tts.create({
   563→        input: script.slice(0, 1024),
   564→        voice: VOICE,
   565→        speed: VOICE_SPEEDS[lang],
   566→        response_format: "wav",
   567→        stream: false,
   568→      });
   569→      const arrayBuffer = await response.arrayBuffer();
   570→      const candidate = Buffer.from(new Uint8Array(arrayBuffer));
   571→      if (candidate.length >= 1000) buffer = candidate;
   572→      else throw new Error("TTS returned an empty audio buffer");
   573→    } catch (err) {
   574→      const msg = String((err as Error)?.message ?? err);
   575→      const last = attempt === 3;
   576→      if (last) throw err;
   577→      if (!msg.includes("429") && !msg.includes("Too many")) throw err;
   578→      console.warn(`[intro-voice:${lang}] TTS rate-limited, retrying (attempt ${attempt + 1})`);
   579→    }
   580→  }
   581→  if (!buffer) throw new Error("TTS failed after retries");
   582→  fs.mkdirSync(AUDIO_DIR, { recursive: true });
   583→  const rawPath = path.join(AUDIO_DIR, `intro-voice-${lang}-${Date.now()}.wav`);
   584→  fs.writeFileSync(rawPath, buffer);
   585→
   586→  const audioPath = path.join(AUDIO_DIR, `intro-voice-${lang}.mp3`);
   587→  try {
   588→    const dur = probeDuration(rawPath);
   589→    const bed = await ensureBed(lang, dur);
   590→    const total = dur + 2.2;
   591→    const fadeStart = Math.max(0, total - 0.9);
   592→    execFileSync(
   593→      "ffmpeg",
   594→      [
   595→        "-y", "-v", "error",
   596→        "-i", rawPath,
   597→        "-i", bed,
   598→        "-filter_complex",
   599→        "[0:a]highpass=f=75," +
   600→          "equalizer=f=260:t=q:w=1.2:g=-2.5," +
   601→          "equalizer=f=3300:t=q:w=1.6:g=2.5," +
   602→          "equalizer=f=8500:t=q:w=2:g=1.8," +
   603→          "acompressor=threshold=-21dB:ratio=3:attack=8:release=140:makeup=3," +
   604→          "aresample=48000," +
   605→          "aformat=sample_fmts=fltp:channel_layouts=stereo," +
   606→          "apad=pad_dur=2.2[vc];" +
   607→          "[1:a]aformat=sample_fmts=fltp:channel_layouts=stereo[bc];" +
   608→          "[vc][bc]amix=inputs=2:duration=first:dropout_transition=0:normalize=0[mx];" +
   609→          `[mx]loudnorm=I=-16:TP=-1.5:LRA=11,aresample=48000,afade=t=out:st=${fadeStart.toFixed(2)}:d=0.9[out]`,
   610→        "-map", "[out]",
   611→        "-codec:a", "libmp3lame",
   612→        "-b:a", "160k",
   613→        "-ar", "48000",
   614→        audioPath,
   615→      ],
   616→      { stdio: "ignore", timeout: 120000 },
   617→    );
   618→  } finally {
   619→    try { fs.unlinkSync(rawPath); } catch { /* cleanup */ }
   620→  }
   621→  fs.writeFileSync(path.join(AUDIO_DIR, `intro-voice-${lang}.script.txt`), `${VOICE}\n${script}`);
   622→  return audioPath;
   623→}

   628→/**
   629→ * Resolve the cached intro audio path for a language, regenerating it when
   630→ * the script or voice changed.
   631→ */
   632→export async function getIntroVoicePath(lang: VoiceLang): Promise<string> {
   633→  // Admin-uploaded custom voice (e.g. an ElevenLabs master) always wins.
   634→  const customPath = customPathFor(lang);
   635→  try {
   636→    if (fs.existsSync(customPath) && fs.statSync(customPath).size > 10000) {
   637→      return customPath;
   638→    }
   639→  } catch {
   640→    /* fall through to the TTS pipeline */
   641→  }
   642→
   643→  let profile: Partial<SiteProfile> = {};
   644→  try {
   645→    profile = (await db.siteProfile.findUnique({ where: { id: "main" } })) ?? {};
   646→  } catch {
   647→    /* fall back to defaults */
   648→  }
   649→  const script = buildIntroScript(lang, profile);
   650→  const audioPath = path.join(AUDIO_DIR, `intro-voice-${lang}.mp3`);
   651→  const scriptPath = path.join(AUDIO_DIR, `intro-voice-${lang}.script.txt`);
   652→
   653→  // Valid cache: same voice + same script + file exists
   654→  try {
   655→    const cached = fs.readFileSync(scriptPath, "utf8");
   656→    if (cached === `${VOICE}\n${script}` && fs.existsSync(audioPath)) {
   657→      return audioPath;
   658→    }
   659→  } catch {
   660→    /* no cache yet */
   661→  }
   662→
   663→  const existing = inFlight.get(lang);
   664→  if (existing) return existing;
   665→  const task = generateAudio(lang, script).finally(() => {
   666→    inFlight.delete(lang);
   667→  });
   668→  inFlight.set(lang, task);
   669→  return task;
   670→}
