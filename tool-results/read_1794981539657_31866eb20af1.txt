   159→   159→function UploadCard({ onUploaded }: { onUploaded: () => void }) {
   160→   160→  const { toast } = useToast();
   161→   161→  const fileInputRef = useRef<HTMLInputElement>(null);
   162→   162→  const thumbInputRef = useRef<HTMLInputElement>(null);
   163→   163→
   164→   164→  const [file, setFile] = useState<File | null>(null);
   165→   165→  const [thumb, setThumb] = useState<File | null>(null);
   166→   166→  const [title, setTitle] = useState("");
   167→   167→  const [description, setDescription] = useState("");
   168→   168→  const [category, setCategory] = useState("meta-ads");
   169→   169→  const [featured, setFeatured] = useState(false);
   170→   170→  const [optimize, setOptimize] = useState(true);
   171→   171→  const [dragOver, setDragOver] = useState(false);
   172→   172→  const [stage, setStage] = useState<"idle" | "uploading" | "processing">("idle");
   173→   173→  const [progress, setProgress] = useState(0);
   174→   174→
   175→   175→  const pickFile = (f: File | null) => {
   176→   176→    if (!f) return;
   177→   177→    const ok = /\.(mp4|webm|mov|mkv|m4v)$/i.test(f.name);
   178→   178→    if (!ok) {
   179→   179→      toast({ title: "Unsupported file", description: "Use mp4, webm, mov or mkv.", variant: "destructive" });
   180→   180→      return;
   181→   181→    }
   182→   182→    setFile(f);
   183→   183→    if (!title) setTitle(f.name.replace(/\.[^.]+$/, "").replace(/[_-]+/g, " ").trim());
   184→   184→  };
   185→   185→
   186→   186→  const reset = () => {
   187→   187→    setFile(null);
   188→   188→    setThumb(null);
   189→   189→    setTitle("");
   190→   190→    setDescription("");
   191→   191→    setCategory("meta-ads");
   192→   192→    setFeatured(false);
   193→   193→    setOptimize(true);
   194→   194→    setProgress(0);
   195→   195→    setStage("idle");
   196→   196→  };
   197→   197→
   198→   198→  const submit = async (e: React.FormEvent) => {
   199→   199→    e.preventDefault();
   200→   200→    if (!file) {
   201→   201→      toast({ title: "No video selected", description: "Choose a video file first.", variant: "destructive" });
   202→   202→      return;
   203→   203→    }
   204→   204→    if (!title.trim()) {
   205→   205→      toast({ title: "Title required", description: "Give the project a title.", variant: "destructive" });
   206→   206→      return;
   207→   207→    }
   208→   208→
   209→   209→    const fd = new FormData();
   210→   210→    fd.append("video", file);
   211→   211→    if (thumb) fd.append("thumbnail", thumb);
   212→   212→    fd.append("title", title.trim());
   213→   213→    fd.append("description", description.trim());
   214→   214→    fd.append("category", category);
   215→   215→    fd.append("featured", String(featured));
   216→   216→    fd.append("optimize", String(optimize));
   217→   217→
   218→   218→    setStage("uploading");
   219→   219→    setProgress(0);
   220→   220→
   221→   221→    try {
   222→   222→      await new Promise<void>((resolve, reject) => {
   223→   223→        const xhr = new XMLHttpRequest();
   224→   224→        xhr.open("POST", "/api/admin/videos");
   225→   225→        xhr.upload.onprogress = (e) => {
   226→   226→          if (e.lengthComputable) {
   227→   227→            const pct = Math.round((e.loaded / e.total) * 100);
   228→   228→            setProgress(pct);
   229→   229→            if (pct >= 100) setStage("processing");
   230→   230→          }
   231→   231→        };
   232→   232→        xhr.onload = () => {
   233→   233→          try {
   234→   234→            const data = JSON.parse(xhr.responseText) as { ok: boolean; error?: string; optimized?: boolean };
   235→   235→            if (xhr.status >= 200 && xhr.status < 300 && data.ok) {
   236→   236→              toast({
   237→   237→                title: "Video published",
   238→   238→                description: data.optimized
   239→   239→                  ? "Uploaded, optimized for web and added to the site."
   240→   240→                  : "Uploaded and added to the site.",
   241→   241→              });
   242→   242→              resolve();
   243→   243→            } else {
   244→   244→              reject(new Error(data.error ?? "Upload failed."));
   245→   245→            }
   246→   246→          } catch {
   247→   247→            reject(new Error("Unexpected server response."));
   248→   248→          }
   249→   249→        };
   250→   250→        xhr.onerror = () => reject(new Error("Network error during upload."));
   251→   251→        xhr.send(fd);
   252→   252→      });
   253→   253→      reset();
   254→   254→      onUploaded();
   255→   255→    } catch (err) {
   256→   256→      toast({
   257→   257→        title: "Upload failed",
   258→   258→        description: err instanceof Error ? err.message : "Try again.",
   259→   259→        variant: "destructive",
   260→   260→      });
   261→   261→      setStage("idle");
   262→   262→    }
   263→   263→  };
   264→   264→
   265→   265→  const busy = stage !== "idle";
   266→   266→  const sizeMb = file ? Math.round((file.size / 1048576) * 10) / 10 : 0;
   267→   267→
   268→   268→  return (
   269→   269→    <form
   270→   270→      onSubmit={submit}
   271→   271→      className="rounded-3xl border border-line bg-card/60 p-5 sm:p-6"
   272→   272→      aria-label="Upload video"
   273→   273→    >
   274→   274→      <h2 className="font-display flex items-center gap-2 text-sm font-bold uppercase tracking-wider text-foreground">
   275→   275→        <UploadCloud className="h-4 w-4 text-glow" />
   276→   276→        Upload a video
   277→   277→      </h2>
   278→   278→
   279→   279→      {/* Drop zone */}
   280→   280→      <button
   281→   281→        type="button"
   282→   282→        onClick={() => fileInputRef.current?.click()}
   283→   283→        onDragOver={(e) => {
   284→   284→          e.preventDefault();
   285→   285→          setDragOver(true);
   286→   286→        }}
   287→   287→        onDragLeave={() => setDragOver(false)}
   288→   288→        onDrop={(e) => {
   289→   289→          e.preventDefault();
   290→   290→          setDragOver(false);
   291→   291→          pickFile(e.dataTransfer.files?.[0] ?? null);
   292→   292→        }}
   293→   293→        className={cn(
   294→   294→          "mt-4 flex w-full flex-col items-center justify-center gap-2 rounded-2xl border-2 border-dashed p-7 text-center transition",
   295→   295→          dragOver ? "border-electric bg-electric/10" : "border-line bg-surface/40 hover:border-electric/60"
   296→   296→        )}
   297→   297→      >
   298→   298→        <Film className="h-7 w-7 text-glow" />
   299→   299→        {file ? (
   300→   300→          <>
   301→   301→            <p className="max-w-full truncate text-sm font-semibold text-foreground">{file.name}</p>
   302→   302→            <p className="text-[11px] text-muted-foreground">{sizeMb} MB — click to replace</p>
   303→   303→          </>
   304→   304→        ) : (
   305→   305→          <>
   306→   306→            <p className="text-sm font-semibold text-foreground">Tap to choose a video — or drop it here</p>
   307→   307→            <p className="text-[11px] text-muted-foreground">mp4 / webm / mov / mkv — up to 600 MB</p>
   308→   308→          </>
   309→   309→        )}
   310→   310→      </button>
   311→   311→      <input
   312→   312→        ref={fileInputRef}
   313→   313→        type="file"
   314→   314→        accept="video/mp4,video/webm,video/quicktime,video/x-matroska"
   315→   315→        className="hidden"
   316→   316→        onChange={(e) => pickFile(e.target.files?.[0] ?? null)}
   317→   317→      />
   318→   318→
   319→   319→      {/* Fields */}
   320→   320→      <div className="mt-4 grid gap-4 sm:grid-cols-2">
   321→   321→        <div className="space-y-1.5 sm:col-span-2">
   322→   322→          <Label htmlFor="v-title">Title *</Label>
   323→   323→          <Input
   324→   324→            id="v-title"
   325→   325→            value={title}
   326→   326→            onChange={(e) => setTitle(e.target.value)}
   327→   327→            placeholder="e.g. Client Name — Product Ad"
   328→   328→            maxLength={120}
   329→   329→          />
   330→   330→        </div>
   331→   331→        <div className="space-y-1.5 sm:col-span-2">
   332→   332→          <Label htmlFor="v-desc">Description</Label>
   333→   333→          <Textarea
   334→   334→            id="v-desc"
   335→   335→            value={description}
   336→   336→            onChange={(e) => setDescription(e.target.value)}
   337→   337→            placeholder="One or two lines about the edit — shown on the card."
   338→   338→            rows={3}
   339→   339→            maxLength={400}
   340→   340→          />
   341→   341→        </div>
   342→   342→        <div className="space-y-1.5">
   343→   343→          <Label>Category</Label>
   344→   344→          <Select value={category} onValueChange={setCategory}>
   345→   345→            <SelectTrigger aria-label="Category">

   347→   347→            </SelectTrigger>
   348→   348→            <SelectContent>
   349→   349→              {CATEGORIES.map((c) => (
   350→   350→                <SelectItem key={c.id} value={c.id}>
   351→   351→                  {c.label}
   352→   352→                </SelectItem>

   354→   354→            </SelectContent>

   362→   362→          <input
