    89→        method: "POST",

   159→function UploadCard({ onUploaded }: { onUploaded: () => void }) {
   160→  const { toast } = useToast();
   161→  const fileInputRef = useRef<HTMLInputElement>(null);
   162→  const thumbInputRef = useRef<HTMLInputElement>(null);
   163→
   164→  const [file, setFile] = useState<File | null>(null);
   165→  const [thumb, setThumb] = useState<File | null>(null);
   166→  const [title, setTitle] = useState("");
   167→  const [description, setDescription] = useState("");
   168→  const [category, setCategory] = useState("meta-ads");
   169→  const [featured, setFeatured] = useState(false);
   170→  const [optimize, setOptimize] = useState(true);
   171→  const [dragOver, setDragOver] = useState(false);
   172→  const [stage, setStage] = useState<"idle" | "uploading" | "processing">("idle");
   173→  const [progress, setProgress] = useState(0);
   174→
   175→  const pickFile = (f: File | null) => {
   176→    if (!f) return;
   177→    const ok = /\.(mp4|webm|mov|mkv|m4v)$/i.test(f.name);
   178→    if (!ok) {
   179→      toast({ title: "Unsupported file", description: "Use mp4, webm, mov or mkv.", variant: "destructive" });
   180→      return;
   181→    }
   182→    setFile(f);
   183→    if (!title) setTitle(f.name.replace(/\.[^.]+$/, "").replace(/[_-]+/g, " ").trim());
   184→  };
   185→
   186→  const reset = () => {
   187→    setFile(null);
   188→    setThumb(null);
   189→    setTitle("");
   190→    setDescription("");
   191→    setCategory("meta-ads");
   192→    setFeatured(false);
   193→    setOptimize(true);
   194→    setProgress(0);
   195→    setStage("idle");
   196→  };
   197→
   198→  const submit = async (e: React.FormEvent) => {
   199→    e.preventDefault();
   200→    if (!file) {
   201→      toast({ title: "No video selected", description: "Choose a video file first.", variant: "destructive" });
   202→      return;
   203→    }
   204→    if (!title.trim()) {
   205→      toast({ title: "Title required", description: "Give the project a title.", variant: "destructive" });
   206→      return;
   207→    }
   208→
   209→    const fd = new FormData();
   210→    fd.append("video", file);
   211→    if (thumb) fd.append("thumbnail", thumb);
   212→    fd.append("title", title.trim());
   213→    fd.append("description", description.trim());
   214→    fd.append("category", category);
   215→    fd.append("featured", String(featured));
   216→    fd.append("optimize", String(optimize));
   217→
   218→    setStage("uploading");
   219→    setProgress(0);
   220→
   221→    try {
   222→      await new Promise<void>((resolve, reject) => {
   223→        const xhr = new XMLHttpRequest();
   224→        xhr.open("POST", "/api/admin/videos");
   225→        xhr.upload.onprogress = (e) => {
   226→          if (e.lengthComputable) {
   227→            const pct = Math.round((e.loaded / e.total) * 100);
   228→            setProgress(pct);
   229→            if (pct >= 100) setStage("processing");
   230→          }
   231→        };
   232→        xhr.onload = () => {
   233→          try {
   234→            const data = JSON.parse(xhr.responseText) as { ok: boolean; error?: string; optimized?: boolean };
   235→            if (xhr.status >= 200 && xhr.status < 300 && data.ok) {
   236→              toast({
   237→                title: "Video published",
   238→                description: data.optimized
   239→                  ? "Uploaded, optimized for web and added to the site."
   240→                  : "Uploaded and added to the site.",
   241→              });
   242→              resolve();
   243→            } else {
   244→              reject(new Error(data.error ?? "Upload failed."));
   245→            }
   246→          } catch {
   247→            reject(new Error("Unexpected server response."));
   248→          }
   249→        };
   250→        xhr.onerror = () => reject(new Error("Network error during upload."));
   251→        xhr.send(fd);
   252→      });
   253→      reset();
   254→      onUploaded();
   255→    } catch (err) {
   256→      toast({
   257→        title: "Upload failed",
   258→        description: err instanceof Error ? err.message : "Try again.",
   259→        variant: "destructive",
   260→      });
   261→      setStage("idle");
   262→    }
   263→  };
   264→
   265→  const busy = stage !== "idle";
   266→  const sizeMb = file ? Math.round((file.size / 1048576) * 10) / 10 : 0;
   267→
   268→  return (
   269→    <form
   270→      onSubmit={submit}
   271→      className="rounded-3xl border border-line bg-card/60 p-5 sm:p-6"
   272→      aria-label="Upload video"
   273→    >
   274→      <h2 className="font-display flex items-center gap-2 text-sm font-bold uppercase tracking-wider text-foreground">
   275→        <UploadCloud className="h-4 w-4 text-glow" />
   276→        Upload a video
   277→      </h2>
   278→
   279→      {/* Drop zone */}
   280→      <button
   281→        type="button"
   282→        onClick={() => fileInputRef.current?.click()}
   283→        onDragOver={(e) => {
   284→          e.preventDefault();
   285→          setDragOver(true);
   286→        }}
   287→        onDragLeave={() => setDragOver(false)}
   288→        onDrop={(e) => {
   289→          e.preventDefault();
   290→          setDragOver(false);
   291→          pickFile(e.dataTransfer.files?.[0] ?? null);
   292→        }}
   293→        className={cn(
   294→          "mt-4 flex w-full flex-col items-center justify-center gap-2 rounded-2xl border-2 border-dashed p-7 text-center transition",
   295→          dragOver ? "border-electric bg-electric/10" : "border-line bg-surface/40 hover:border-electric/60"
   296→        )}
   297→      >
   298→        <Film className="h-7 w-7 text-glow" />
   299→        {file ? (
   300→          <>
   301→            <p className="max-w-full truncate text-sm font-semibold text-foreground">{file.name}</p>
   302→            <p className="text-[11px] text-muted-foreground">{sizeMb} MB — click to replace</p>
   303→          </>
   304→        ) : (
   305→          <>
   306→            <p className="text-sm font-semibold text-foreground">Tap to choose a video — or drop it here</p>
   307→            <p className="text-[11px] text-muted-foreground">mp4 / webm / mov / mkv — up to 600 MB</p>
   308→          </>
   309→        )}
   310→      </button>
   311→      <input
   312→        ref={fileInputRef}
   313→        type="file"
   314→        accept="video/mp4,video/webm,video/quicktime,video/x-matroska"
   315→        className="hidden"
   316→        onChange={(e) => pickFile(e.target.files?.[0] ?? null)}
   317→      />
   318→
   319→      {/* Fields */}
   320→      <div className="mt-4 grid gap-4 sm:grid-cols-2">
   321→        <div className="space-y-1.5 sm:col-span-2">
   322→          <Label htmlFor="v-title">Title *</Label>
   323→          <Input
   324→            id="v-title"
   325→            value={title}
   326→            onChange={(e) => setTitle(e.target.value)}
   327→            placeholder="e.g. Client Name — Product Ad"
   328→            maxLength={120}
   329→          />
   330→        </div>
   331→        <div className="space-y-1.5 sm:col-span-2">
   332→          <Label htmlFor="v-desc">Description</Label>
   333→          <Textarea
   334→            id="v-desc"
   335→            value={description}
   336→            onChange={(e) => setDescription(e.target.value)}
   337→            placeholder="One or two lines about the edit — shown on the card."
   338→            rows={3}
   339→            maxLength={400}
   340→          />
   341→        </div>
   342→        <div className="space-y-1.5">
   343→          <Label>Category</Label>
   344→          <Select value={category} onValueChange={setCategory}>
   345→            <SelectTrigger aria-label="Category">
   346→              <SelectValue placeholder="Pick a category" />
   347→            </SelectTrigger>
   348→            <SelectContent>
   349→              {CATEGORIES.map((c) => (
   350→                <SelectItem key={c.id} value={c.id}>
   351→                  {c.label}
   352→                </SelectItem>
   353→              ))}
   354→            </SelectContent>
   355→          </Select>
   356→        </div>
   357→        <div className="space-y-1.5">
   358→          <Label>Custom thumbnail (optional)</Label>
   359→          <Button type="button" variant="outline" onClick={() => thumbInputRef.current?.click()} className="w-full">
   360→            {thumb ? thumb.name.slice(0, 28) : "Choose image…"}
   361→          </Button>
   362→          <input
   363→            ref={thumbInputRef}
   364→            type="file"
   365→            accept="image/*"
   366→            className="hidden"
   367→            onChange={(e) => setThumb(e.target.files?.[0] ?? null)}
   368→          />
   369→        </div>
   370→
   371→        <div className="flex items-center justify-between gap-3 rounded-xl border border-line bg-surface/40 px-4 py-3 sm:col-span-1">
   372→          <div>
   373→            <p className="text-sm font-semibold text-foreground">Feature as showreel</p>
   374→            <p className="text-[11px] text-muted-foreground">Pin it to the Showreel section</p>
   375→          </div>
   376→          <Switch checked={featured} onCheckedChange={setFeatured} aria-label="Feature as showreel" />
   377→        </div>
   378→        <div className="flex items-center justify-between gap-3 rounded-xl border border-line bg-surface/40 px-4 py-3 sm:col-span-1">
   379→          <div>
   380→            <p className="text-sm font-semibold text-foreground">Optimize for web</p>
   381→            <p className="text-[11px] text-muted-foreground">Re-encode to a fast-loading MP4</p>
   382→          </div>
   383→          <Switch checked={optimize} onCheckedChange={setOptimize} aria-label="Optimize for web" />
   384→        </div>
   385→      </div>
   386→
   387→      {/* Progress */}
   388→      {busy ? (
   389→        <div className="mt-4">
   390→          <div className="flex items-center justify-between text-[11px] text-muted-foreground">
   391→            <span>{stage === "uploading" ? "Uploading…" : "Processing — encoding for web…"}</span>
   392→            <span className="tabular-nums">{stage === "uploading" ? `${progress}%` : ""}</span>
   393→          </div>
   394→          <div className="mt-1.5 h-2 overflow-hidden rounded-full bg-surface">
   395→            <div
   396→              className="h-full rounded-full bg-electric transition-all duration-300"
   397→              style={{ width: stage === "uploading" ? `${progress}%` : "100%" }}
   398→            />
   399→          </div>
   400→          {stage === "processing" ? (
   401→            <p className="mt-2 flex items-center gap-1.5 text-[11px] text-muted-foreground">
   402→              <Loader2 className="h-3 w-3 animate-spin" />
   403→              Large files can take a minute — keep this tab open.
   404→            </p>
   405→          ) : null}
   406→        </div>
   407→      ) : null}
   408→
   409→      <div className="mt-5 flex gap-3">
   410→        <Button type="submit" disabled={busy} className="flex-1 gap-2 bg-electric hover:bg-glow">
   411→          {busy ? <Loader2 className="h-4 w-4 animate-spin" /> : <UploadCloud className="h-4 w-4" />}
   412→          {busy ? "Working…" : "Publish video"}
   413→        </Button>
   414→        {file && !busy ? (
   415→          <Button type="button" variant="outline" onClick={reset}>
   416→            Clear
   417→          </Button>
   418→        ) : null}
   419→      </div>
   420→    </form>
   421→  );
   422→}

   539→  useEffect(() => {
   540→    void checkSession();
   541→  }, [checkSession]);
   542→
   543→  const loadVideos = useCallback(async () => {
   544→    setLoadingList(true);
   545→    try {
   546→      const res = await fetch("/api/admin/videos");
   547→      if (res.status === 401) {
   548→        setAuthed(false);
   549→        return;
   550→      }
   551→      const data = (await res.json()) as { videos: AdminVideo[] };
   552→      setVideos(data.videos ?? []);
   553→    } catch {
   554→      toast({ title: "Could not load library", variant: "destructive" });
   555→    } finally {
   556→      setLoadingList(false);
   557→    }
   558→  }, [toast]);
   559→
   560→  useEffect(() => {
   561→    if (authed) void loadVideos();
   562→  }, [authed, loadVideos]);
   563→
   564→  const logout = async () => {
   565→    await fetch("/api/admin/logout", { method: "POST" });
   566→    setAuthed(false);
   567→    setVideos([]);
   568→  };
   569→
   570→  const saveEdit = async () => {
   571→    if (!editing) return;
   572→    if (!editTitle.trim()) {
   573→      toast({ title: "Title required", variant: "destructive" });
   574→      return;
   575→    }
   576→    setEditBusy(true);
   577→    try {
   578→      const res = await fetch(`/api/admin/videos/${editing.id}`, {
   579→        method: "PUT",
   580→        headers: { "Content-Type": "application/json" },
   581→        body: JSON.stringify({
   582→          title: editTitle,
   583→          description: editDescription,
   584→          category: editCategory,
   585→        }),
   586→      });
   587→      const data = (await res.json()) as { ok: boolean; error?: string };
   588→      if (data.ok) {
   589→        toast({ title: "Saved", description: "Video updated on the site." });
   590→        setEditing(null);
   591→        void loadVideos();
   592→      } else {
   593→        toast({ title: "Update failed", description: data.error, variant: "destructive" });
   594→      }
   595→    } catch {
   596→      toast({ title: "Update failed", variant: "destructive" });
   597→    } finally {
   598→      setEditBusy(false);
   599→    }
   600→  };
   601→
   602→  const toggleFeatured = async (video: AdminVideo) => {
   603→    try {
   604→      const res = await fetch(`/api/admin/videos/${video.id}`, {
   605→        method: "PUT",
   606→        headers: { "Content-Type": "application/json" },
   607→        body: JSON.stringify({ featured: !video.featured }),
   608→      });
   609→      const data = (await res.json()) as { ok: boolean; error?: string };
   610→      if (data.ok) {
   611→        toast({
   612→          title: video.featured ? "Removed from showreel" : "Pinned as showreel",
   613→          description: video.featured ? undefined : "It now plays in the Showreel section.",
   614→        });
   615→        void loadVideos();
   616→      } else {
   617→        toast({ title: "Update failed", description: data.error, variant: "destructive" });
   618→      }
   619→    } catch {
   620→      toast({ title: "Update failed", variant: "destructive" });
   621→    }
   622→  };
   623→
   624→  const confirmDelete = async () => {
   625→    if (!deleting) return;
   626→    setDeleteBusy(true);
   627→    try {
   628→      const res = await fetch(`/api/admin/videos/${deleting.id}`, { method: "DELETE" });
   629→      const data = (await res.json()) as { ok: boolean; error?: string };
   630→      if (data.ok) {
   631→        toast({ title: "Deleted", description: `"${deleting.title}" was removed from the site.` });
   632→        setDeleting(null);
   633→        void loadVideos();
   634→      } else {
   635→        toast({ title: "Delete failed", description: data.error, variant: "destructive" });
   636→      }
   637→    } catch {
   638→      toast({ title: "Delete failed", variant: "destructive" });
   639→    } finally {
   640→      setDeleteBusy(false);
   641→    }
   642→  };
   643→
   644→  if (authed === null) {
   645→    return (
   646→      <div className="flex min-h-screen items-center justify-center">
   647→        <Loader2 className="h-6 w-6 animate-spin text-electric" />
   648→      </div>
   649→    );
   650→  }
   651→
   652→  if (!authed) {
   653→    return (
   654→      <LoginScreen
   655→        onSuccess={() => setAuthed(true)}
   656→        usingDefaultPassword={usingDefaultPassword}
   657→      />
   658→    );
   659→  }
   660→
   661→  return (
   662→    <div className="min-h-screen px-4 py-8 sm:px-6 sm:py-10">
   663→      <div className="mx-auto max-w-5xl">
   664→        {/* Header */}
   665→        <header className="flex flex-wrap items-center justify-between gap-3">
   666→          <div className="flex items-center gap-3">
   667→            <span className="flex h-10 w-10 items-center justify-center rounded-xl border border-electric/50 bg-electric/10">
   668→              <Clapperboard className="h-5 w-5 text-glow" />
   669→            </span>
   670→            <div>
   671→              <p className="font-display text-sm font-bold tracking-[0.18em] text-foreground">
   672→                UBAID <span className="text-glow">EDITS</span>
   673→              </p>
   674→              <p className="text-[11px] font-semibold uppercase tracking-[0.24em] text-muted-foreground">
   675→                Admin Panel
   676→              </p>
   677→            </div>
   678→          </div>
   679→          <div className="flex items-center gap-2">
   680→            <a
   681→              href="/"
   682→              className="rounded-lg border border-line px-3.5 py-2 text-xs font-semibold text-muted-foreground transition hover:border-electric hover:text-foreground"
   683→            >
   684→              View site
   685→            </a>
   686→            <Button variant="outline" onClick={logout} className="gap-2">
   687→              <LogOut className="h-4 w-4" />
   688→              Logout
   689→            </Button>
   690→          </div>
   691→        </header>
   692→
   693→        <main className="mt-8 grid gap-6 lg:grid-cols-[1fr_1.1fr]">
   694→          <UploadCard onUploaded={() => void loadVideos()} />
   695→
   696→          {/* Library */}
   697→          <section aria-label="Published videos" className="rounded-3xl border border-line bg-card/40 p-5 sm:p-6">
   698→            <div className="flex items-center justify-between gap-3">
   699→              <h2 className="font-display flex items-center gap-2 text-sm font-bold uppercase tracking-wider text-foreground">
   700→                <Film className="h-4 w-4 text-glow" />
   701→                Published videos
   702→                <Badge variant="outline" className="border-line text-[10px] text-muted-foreground">
   703→                  {videos.length}
   704→                </Badge>
   705→              </h2>
   706→              <button
   707→                onClick={() => void loadVideos()}
   708→                aria-label="Refresh list"
   709→                className="rounded-lg border border-line p-2 text-muted-foreground transition hover:border-electric hover:text-foreground"
   710→              >
   711→                <RefreshCw className={cn("h-4 w-4", loadingList && "animate-spin")} />
   712→              </button>
   713→            </div>
   714→
   715→            <div className="mt-4 max-h-[720px] space-y-3 overflow-y-auto pe-1">
   716→              {loadingList && videos.length === 0 ? (
   717→                <div className="flex items-center justify-center py-10">
   718→                  <Loader2 className="h-5 w-5 animate-spin text-electric" />
   719→                </div>
   720→              ) : videos.length === 0 ? (
   721→                <div className="flex flex-col items-center gap-2 rounded-2xl border border-dashed border-line p-8 text-center">
   722→                  <Film className="h-6 w-6 text-muted-foreground/50" />
   723→                  <p className="text-sm text-muted-foreground">
   724→                    No videos published yet — upload the first one.
   725→                  </p>
   726→                </div>
   727→              ) : (
   728→                videos.map((v) => (
   729→                  <LibraryRow
   730→                    key={v.id}
   731→                    video={v}
   732→                    onEdit={() => {
   733→                      setEditing(v);
   734→                      setEditTitle(v.title);
   735→                      setEditDescription(v.description);
   736→                      setEditCategory(v.category);
   737→                    }}
   738→                    onDelete={() => setDeleting(v)}
   739→                    onToggleFeatured={() => void toggleFeatured(v)}
   740→                  />
   741→                ))
   742→              )}
   743→            </div>
   744→          </section>
   745→        </main>
   746→      </div>
   747→
   748→      {/* Edit dialog */}
   749→      <Dialog open={editing !== null} onOpenChange={(v) => !v && setEditing(null)}>
   750→        <DialogContent className="border-line bg-card p-5 sm:p-6">
   751→          <DialogHeader>
   752→            <DialogTitle className="font-display text-foreground">Edit video</DialogTitle>
   753→          </DialogHeader>
   754→          <div className="space-y-4">
   755→            <div className="space-y-1.5">
   756→              <Label htmlFor="edit-title">Title</Label>
   757→              <Input id="edit-title" value={editTitle} onChange={(e) => setEditTitle(e.target.value)} maxLength={120} />
   758→            </div>
   759→            <div className="space-y-1.5">
   760→              <Label htmlFor="edit-desc">Description</Label>
   761→              <Textarea
   762→                id="edit-desc"
   763→                value={editDescription}
   764→                onChange={(e) => setEditDescription(e.target.value)}
   765→                rows={3}
   766→                maxLength={400}
   767→              />
   768→            </div>
   769→            <div className="space-y-1.5">
   770→              <Label>Category</Label>
   771→              <Select value={editCategory} onValueChange={setEditCategory}>
   772→                <SelectTrigger aria-label="Category">
   773→                  <SelectValue />
   774→                </SelectTrigger>
   775→                <SelectContent>
   776→                  {CATEGORIES.map((c) => (
   777→                    <SelectItem key={c.id} value={c.id}>
   778→                      {c.label}
   779→                    </SelectItem>
   780→                  ))}
   781→                </SelectContent>
   782→              </Select>
   783→            </div>
   784→            <div className="flex justify-end gap-3 pt-1">
   785→              <Button variant="outline" onClick={() => setEditing(null)}>
   786→                Cancel
   787→              </Button>
   788→              <Button onClick={saveEdit} disabled={editBusy} className="gap-2 bg-electric hover:bg-glow">
   789→                {editBusy ? <Loader2 className="h-4 w-4 animate-spin" /> : null}
   790→                Save changes
   791→              </Button>
   792→            </div>
   793→          </div>
   794→        </DialogContent>
   795→      </Dialog>
   796→
   797→      {/* Delete confirm */}
   798→      <Dialog open={deleting !== null} onOpenChange={(v) => !v && setDeleting(null)}>
   799→        <DialogContent className="border-line bg-card p-5 sm:p-6">
   800→          <DialogHeader>
   801→            <DialogTitle className="font-display text-foreground">Delete video?</DialogTitle>
   802→          </DialogHeader>
   803→          <p className="text-sm text-muted-foreground">
   804→            &ldquo;{deleting?.title}&rdquo; will be removed from the site and its files deleted. This
   805→            cannot be undone.
   806→          </p>
   807→          <div className="flex justify-end gap-3 pt-2">
   808→            <Button variant="outline" onClick={() => setDeleting(null)}>
   809→              Cancel
   810→            </Button>
   811→            <Button onClick={confirmDelete} disabled={deleteBusy} variant="destructive" className="gap-2">
   812→              {deleteBusy ? <Loader2 className="h-4 w-4 animate-spin" /> : <Trash2 className="h-4 w-4" />}
   813→              Delete
   814→            </Button>
   815→          </div>
   816→        </DialogContent>
   817→      </Dialog>
   818→    </div>
   819→  );
   820→}
