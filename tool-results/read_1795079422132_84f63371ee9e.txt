    68→    68→export function intToWords(n: number): string {
    69→    69→  if (!Number.isFinite(n) || n < 0) return String(n);
    70→    70→  const i = Math.round(n);
    71→    71→  if (i < 20) return ONES[i];
    72→    72→  if (i < 100) {
    73→    73→    const t = Math.floor(i / 10);
    74→    74→    const r = i % 10;
    75→    75→    return r ? `${TENS[t]} ${ONES[r]}` : TENS[t];
    76→    76→  }
    77→    77→  if (i < 1000) {
    78→    78→    const h = Math.floor(i / 100);
    79→    79→    const r = i % 100;
    80→    80→    return r ? `${ONES[h]} hundred and ${intToWords(r)}` : `${ONES[h]} hundred`;
    81→    81→  }

    83→    83→  const r = i % 1000;
    84→    84→  const tail = r ? (r < 100 ? " and " : " ") + intToWords(r) : "";
    85→    85→  return `${intToWords(th)} thousand${tail}`;

   344→   344→/* --------------------------------------------------------------------------
   345→   345→ * Ambient music bed (synthesized, no assets needed) — per language duration
   346→   346→ * ------------------------------------------------------------------------ */
   347→   347→
   348→   348→/**
   349→   349→ * Write a warm ambient pad as a 48 kHz stereo WAV — five slow-evolving sine
   350→   350→ * pairs with gentle detune (width), a soft attack and a long fade-out.
   351→   351→ */
   352→   352→function synthesizeBed(wavPath: string, seconds: number): void {
   353→   353→  const SR = 48000;
   354→   354→  const CH = 2;
   355→   355→  const n = Math.max(1, Math.floor(seconds * SR));
   356→   356→  // [frequency, relative amplitude] — A2, E3, A3, B3, E4 (Am add9, warm)
   357→   357→  const NOTES: Array<[number, number]> = [
   358→   358→    [110.0, 1.0],
   359→   359→    [164.81, 0.72],
   360→   360→    [220.0, 0.5],
   361→   361→    [246.94, 0.4],
   362→   362→    [329.63, 0.28],
   363→   363→  ];
   364→   364→  const l = new Float32Array(n);
   365→   365→  const r = new Float32Array(n);
   366→   366→  for (const [f, a] of NOTES) {
   367→   367→    const fl = f * 0.9986;
   368→   368→    const fr = f * 1.0014;
   369→   369→    const pl = f * 1.7;
   370→   370→    const pr = f * 3.1;
   371→   371→    for (let i = 0; i < n; i++) {
   372→   372→      const t = i / SR;
   373→   373→      l[i] += a * Math.sin(2 * Math.PI * fl * t + pl);
   374→   374→      r[i] += a * Math.sin(2 * Math.PI * fr * t + pr);
   375→   375→    }
   376→   376→  }
   377→   377→  const ATTACK = 2.4;
   378→   378→  const RELEASE = 4.0;
   379→   379→  let peak = 1e-9;
   380→   380→  for (let i = 0; i < n; i++) {
   381→   381→    const t = i / SR;
   382→   382→    let env = t < ATTACK ? (t / ATTACK) * (t / ATTACK) * (3 - 2 * (t / ATTACK)) : 1;
   383→   383→    const relStart = Math.max(0, seconds - RELEASE);
   384→   384→    if (t > relStart) {
   385→   385→      const p = Math.min(1, (t - relStart) / RELEASE);
   386→   386→      env *= 0.5 * (1 + Math.cos(Math.PI * p));
   387→   387→    }
   388→   388→    env *= 1 + 0.15 * Math.sin(2 * Math.PI * 0.13 * t);
   389→   389→    env *= 1 + 0.12 * Math.sin(2 * Math.PI * 0.09 * t + 1.2);
   390→   390→    l[i] *= env;
   391→   391→    r[i] *= env;
   392→   392→    const ap = Math.abs(l[i]);
   393→   393→    const bp = Math.abs(r[i]);
   394→   394→    if (ap > peak) peak = ap;
   395→   395→    if (bp > peak) peak = bp;
   396→   396→  }
   397→   397→  const scale = BED_PEAK / peak;
   398→   398→  const dataSize = n * CH * 2;
   399→   399→  const buf = Buffer.alloc(44 + dataSize);
   400→   400→  buf.write("RIFF", 0);
   401→   401→  buf.writeUInt32LE(36 + dataSize, 4);
   402→   402→  buf.write("WAVE", 8);
   403→   403→  buf.write("fmt ", 12);
   404→   404→  buf.writeUInt32LE(16, 16);
   405→   405→  buf.writeUInt16LE(1, 20);
   406→   406→  buf.writeUInt16LE(CH, 22);
   407→   407→  buf.writeUInt32LE(SR, 24);
   408→   408→  buf.writeUInt32LE(SR * CH * 2, 28);
   409→   409→  buf.writeUInt16LE(CH * 2, 32);
   410→   410→  buf.writeUInt16LE(16, 34);
   411→   411→  buf.write("data", 36);
   412→   412→  buf.writeUInt32LE(dataSize, 40);
   413→   413→  for (let i = 0; i < n; i++) {
   414→   414→    const lv = Math.max(-1, Math.min(1, l[i] * scale));
   415→   415→    const rv = Math.max(-1, Math.min(1, r[i] * scale));
   416→   416→    buf.writeInt16LE(Math.round(lv * 32767), 44 + i * 4);
   417→   417→    buf.writeInt16LE(Math.round(rv * 32767), 44 + i * 4 + 2);
   418→   418→  }
   419→   419→  fs.writeFileSync(wavPath, buf);
   420→   420→  fs.writeFileSync(wavPath.replace(/\.wav$/, ".duration.txt"), seconds.toFixed(2));
   421→   421→}

   423→   423→async function ensureBed(lang: VoiceLang, voiceSeconds: number): Promise<string> {
   424→   424→  const bedPath = path.join(AUDIO_DIR, `intro-bed-${lang}.wav`);

   426→   426→  const need = voiceSeconds + 3.0; // voice + 2.2s pad + tail margin

   430→   430→      return bedPath;
   431→   431→    }
   432→   432→  } catch {
   433→   433→    /* no cache yet */
   434→   434→  }
   435→   435→  synthesizeBed(bedPath, need);
   436→   436→  return bedPath;
   437→   437→}

   552→   552→/** TTS → EQ/compression → mix subtle bed → loudnorm (-16 LUFS) → MP3. */
   553→   553→async function generateAudio(lang: VoiceLang, script: string): Promise<string> {
   554→   554→  const ZAI = (await import("z-ai-web-dev-sdk")).default;
   555→   555→  const zai = await ZAI.create();
   556→   556→
   557→   557→  // Retry with backoff — a throttled TTS API shouldn't fail the visitor's click.
   558→   558→  let buffer: Buffer | null = null;
   559→   559→  for (let attempt = 0; attempt < 4 && !buffer; attempt++) {
   560→   560→    if (attempt > 0) await new Promise((r) => setTimeout(r, 15000 * attempt));
   561→   561→    try {
   562→   562→      const response = await zai.audio.tts.create({
   563→   563→        input: script.slice(0, 1024),
   564→   564→        voice: VOICE,
   565→   565→        speed: VOICE_SPEEDS[lang],
   566→   566→        response_format: "wav",
   567→   567→        stream: false,
   568→   568→      });
   569→   569→      const arrayBuffer = await response.arrayBuffer();
   570→   570→      const candidate = Buffer.from(new Uint8Array(arrayBuffer));
   571→   571→      if (candidate.length >= 1000) buffer = candidate;
   572→   572→      else throw new Error("TTS returned an empty audio buffer");
   573→   573→    } catch (err) {
   574→   574→      const msg = String((err as Error)?.message ?? err);
   575→   575→      const last = attempt === 3;
   576→   576→      if (last) throw err;
   577→   577→      if (!msg.includes("429") && !msg.includes("Too many")) throw err;
   578→   578→      console.warn(`[intro-voice:${lang}] TTS rate-limited, retrying (attempt ${attempt + 1})`);
   579→   579→    }
   580→   580→  }
   581→   581→  if (!buffer) throw new Error("TTS failed after retries");
   582→   582→  fs.mkdirSync(AUDIO_DIR, { recursive: true });
   583→   583→  const rawPath = path.join(AUDIO_DIR, `intro-voice-${lang}-${Date.now()}.wav`);
   584→   584→  fs.writeFileSync(rawPath, buffer);
   585→   585→
   586→   586→  const audioPath = path.join(AUDIO_DIR, `intro-voice-${lang}.mp3`);
   587→   587→  try {
   588→   588→    const dur = probeDuration(rawPath);
   589→   589→    const bed = await ensureBed(lang, dur);
   590→   590→    const total = dur + 2.2;
   591→   591→    const fadeStart = Math.max(0, total - 0.9);
   592→   592→    execFileSync(
   593→   593→      "ffmpeg",
   594→   594→      [
   595→   595→        "-y", "-v", "error",
   596→   596→        "-i", rawPath,
   597→   597→        "-i", bed,
   598→   598→        "-filter_complex",
   599→   599→        "[0:a]highpass=f=75," +
   600→   600→          "equalizer=f=260:t=q:w=1.2:g=-2.5," +
   601→   601→          "equalizer=f=3300:t=q:w=1.6:g=2.5," +
   602→   602→          "equalizer=f=8500:t=q:w=2:g=1.8," +
   603→   603→          "acompressor=threshold=-21dB:ratio=3:attack=8:release=140:makeup=3," +
   604→   604→          "aresample=48000," +
   605→   605→          "aformat=sample_fmts=fltp:channel_layouts=stereo," +
   606→   606→          "apad=pad_dur=2.2[vc];" +
   607→   607→          "[1:a]aformat=sample_fmts=fltp:channel_layouts=stereo[bc];" +
   608→   608→          "[vc][bc]amix=inputs=2:duration=first:dropout_transition=0:normalize=0[mx];" +
   609→   609→          `[mx]loudnorm=I=-16:TP=-1.5:LRA=11,aresample=48000,afade=t=out:st=${fadeStart.toFixed(2)}:d=0.9[out]`,
   610→   610→        "-map", "[out]",
   611→   611→        "-codec:a", "libmp3lame",
   612→   612→        "-b:a", "160k",
   613→   613→        "-ar", "48000",
   614→   614→        audioPath,
   615→   615→      ],
   616→   616→      { stdio: "ignore", timeout: 120000 },
   617→   617→    );
   618→   618→  } finally {
   619→   619→    try { fs.unlinkSync(rawPath); } catch { /* cleanup */ }
   620→   620→  }
   621→   621→  fs.writeFileSync(path.join(AUDIO_DIR, `intro-voice-${lang}.script.txt`), `${VOICE}\n${script}`);
   622→   622→  return audioPath;
   623→   623→}

   628→   628→/**
   629→   629→ * Resolve the cached intro audio path for a language, regenerating it when
   630→   630→ * the script or voice changed.

   632→   632→export async function getIntroVoicePath(lang: VoiceLang): Promise<string> {
   633→   633→  // Admin-uploaded custom voice (e.g. an ElevenLabs master) always wins.
   634→   634→  const customPath = customPathFor(lang);
   635→   635→  try {
   636→   636→    if (fs.existsSync(customPath) && fs.statSync(customPath).size > 10000) {
   637→   637→      return customPath;
   638→   638→    }
   639→   639→  } catch {
   640→   640→    /* fall through to the TTS pipeline */
   641→   641→  }
   642→   642→
   643→   643→  let profile: Partial<SiteProfile> = {};
   644→   644→  try {
   645→   645→    profile = (await db.siteProfile.findUnique({ where: { id: "main" } })) ?? {};
   646→   646→  } catch {
   647→   647→    /* fall back to defaults */
   648→   648→  }
   649→   649→  const script = buildIntroScript(lang, profile);
   650→   650→  const audioPath = path.join(AUDIO_DIR, `intro-voice-${lang}.mp3`);
   651→   651→  const scriptPath = path.join(AUDIO_DIR, `intro-voice-${lang}.script.txt`);
   652→   652→
   653→   653→  // Valid cache: same voice + same script + file exists
   654→   654→  try {
   655→   655→    const cached = fs.readFileSync(scriptPath, "utf8");
   656→   656→    if (cached === `${VOICE}\n${script}` && fs.existsSync(audioPath)) {
   657→   657→      return audioPath;
   658→   658→    }
   659→   659→  } catch {
   660→   660→    /* no cache yet */
   661→   661→  }
   662→   662→
   663→   663→  const existing = inFlight.get(lang);
   664→   664→  if (existing) return existing;
   665→   665→  const task = generateAudio(lang, script).finally(() => {
   666→   666→    inFlight.delete(lang);
   667→   667→  });
   668→   668→  inFlight.set(lang, task);
   669→   669→  return task;
   670→   670→}
